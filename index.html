// ================================================================ // 3. 缓存 // ================================================================ const BILA_BIBLE_CACHE = {}; const BILA_CURRICULUM_CACHE = {}; } let result = String(text); // 去 BOM result = result.replace( /^\uFEFF/, "" ); // 去 HTML result = result.replace( /<[^>]*>/g, "" ); // 去换行 result = result.replace( /[\r\n\t]+/g, " " ); // 多空格 result = result.replace( /\s+/g, " " ); CSV解析 // // 支持： // week,reference,sunday_theme,... // // 并正确处理字段中的逗号和引号。 // ================================================================ function bilaParseCSV(text) { const rows = []; let row = []; let cell = ""; let inQuotes = false; for ( let i = 0; i < text.length; i++ ) { const char = text[i]; const next = text[i + 1]; if (char === '"') { if ( inQuotes && next === '"' ) { cell += '"'; i++; } else { inQuotes = !inQuotes; } continue; } if ( char === "," && !inQuotes ) { row.push( cell.trim() ); cell = ""; continue; } if ( (char === "\n" || char === "\r") && !inQuotes ) { if ( char === "\r" && next === "\n" ) { i++; } row.push( cell.trim() ); if ( row.some( value => value !== "" ) ) { rows.push(row); } row = []; cell = ""; continue; } cell += char; } if ( cell !== "" || row.length ) { row.push( cell.trim() ); if ( row.some( value => value !== "" ) ) { rows.push(row); } } if ( rows.length < 2 ) { return []; } const headers = rows[0].map( header => bilaCleanText(header) ); return rows .slice(1) .map( values => { const obj = {}; headers.forEach( (header, index) => { obj[header] = values[index] || ""; } ); return obj; } ); } // ================================================================ // 6. 加载52周课程 // ================================================================ async function loadCurriculum() { if ( BILA_CURRICULUM_CACHE.courses ) { return BILA_CURRICULUM_CACHE.courses;
